역순 파이프라인: 커스텀 base64 해제 -> 뒤집기 -> nibble swap -> 덧셈 역연산(mod256)
"""

KEY = b"Sn0w"                                        # 서버 비밀 키
CIPHER = "/xlHnM12PzvKS80VLv0GfY2Tghvnne2Y0h023U2jYi5="  # 복호화할 암호문

# 표준 base64 알파벳을 13칸 회전시킨 커스텀 알파벳
STD = "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"
CUSTOM = STD[13:] + STD[:13]
_dec_tab = bytes.maketrans(CUSTOM.encode(), STD.encode())


def custom_b64decode(s: str) -> bytes:
    """커스텀 알파벳을 표준으로 되돌린 뒤 base64 해제"""
    return base64.b64decode(s.encode().translate(_dec_tab))


def nibble_swap(b: bytes) -> bytes:
    """각 바이트의 상위/하위 4비트 교환 (자기역함수)"""
    return bytes(((x << 4) | (x >> 4)) & 0xFF for x in b)


def sub_key(b: bytes, k: bytes) -> bytes:
    """덧셈암호(mod 256)의 역연산"""
    return bytes((b[i] - k[i % len(k)]) & 0xFF for i in range(len(b)))

    raw = nibble_swap(raw)          # 2' 니블 스왑
    return sub_key(raw, key).decode()  # 1' 덧셈 역연산
